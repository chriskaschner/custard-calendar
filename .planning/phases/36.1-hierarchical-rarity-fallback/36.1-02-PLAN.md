**In docs/planner-domain.js -- formatCadenceText function (line 220-228):**

Expand the scope handling to include metro, state, and national:
```javascript
function formatCadenceText(avgGapDays, opts) {
  var days = Math.round(Number(avgGapDays));
  if (!Number.isFinite(days) || days < 2 || days > 365) return '';
  var scope = opts && typeof opts.scope === 'string' ? opts.scope : '';
  var suffix = '';
  if (scope === 'store') suffix = ' at this store';
  if (scope === 'primary') suffix = ' at your store';
  if (scope === 'metro') suffix = ' in your area';
  if (scope === 'state') suffix = ' statewide';
  if (scope === 'national') suffix = ' nationwide';
  return 'Shows up roughly every ' + days + ' days' + suffix;
}
```

**In docs/today-page.js -- renderRarity function (line 326-347):**

Update the cadence text to pass scope from the rarity object:
```javascript
function renderRarity(rarity, flavorName) {
  if (!todayRarity) return;
  if (!rarity) {
    todayRarity.hidden = true;
    return;
  }
  var isSeasonal = CustardPlanner.isSeasonalFlavor(flavorName);
  var html = '';
  if (rarity.label) {
    var cssClass = 'rarity-badge rarity-badge-' + rarity.label.toLowerCase().replace(/\s+/g, '-');
    html += '<span class="' + cssClass + '">' + escapeHtml(rarity.label) + '</span>';
    if (rarity.avg_gap_days && !isSeasonal) {
      var scopeForDisplay = rarity.scope || 'primary';
      html += CustardPlanner.formatCadenceText(rarity.avg_gap_days, { scope: scopeForDisplay });
    }
  }
  if (html) {
    todayRarity.innerHTML = html;
    todayRarity.hidden = false;
  } else {
    todayRarity.hidden = true;
  }
}
```

Key change: Instead of hardcoding "Shows up roughly every N days at your store", use `CustardPlanner.formatCadenceText()` with the scope from the API response. When scope is "store", the text reads "at this store" (D-08 specificity). When scope is null (no scope resolved), default to "primary" which shows "at your store" for backward compatibility.
