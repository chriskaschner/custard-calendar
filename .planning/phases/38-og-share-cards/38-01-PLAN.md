**2. `handleStoreCard` function** -- follows the `handleFlavorCard` pattern (lines 563-621):

```javascript
async function handleStoreCard(storeSlug, env, corsHeaders) {
  // Validate slug is non-empty and looks like a valid slug
  if (!storeSlug || !/^[a-z0-9][a-z0-9-]*$/.test(storeSlug)) {
    return new Response(JSON.stringify({ error: 'Invalid store slug.' }), {
      status: 400,
      headers: { ...corsHeaders, 'Content-Type': 'application/json' },
    });
  }

  // Look up today's flavor from D1 snapshots (best-effort)
  const today = new Date().toISOString().slice(0, 10);
  let flavorName = 'Flavor of the Day';
  const db = env.DB;
  if (db) {
    try {
      const snap = await db.prepare(
        'SELECT flavor FROM snapshots WHERE slug = ? AND date = ? LIMIT 1'
      ).bind(storeSlug, today).first();
      if (snap && snap.flavor) {
        flavorName = snap.flavor;
      }
    } catch {
      // D1 unavailable -- card renders with fallback text
    }
  }

  // Derive store display name from slug: mt-horeb -> Mt Horeb
  const storeName = storeSlug.replace(/-/g, ' ').replace(/\b\w/g, c => c.toUpperCase());

  // Visual properties
  const profile = getFlavorProfile(flavorName);
  const accentColor = BASE_COLORS[profile.base] || '#005696';
  const conePngBase64 = await fetchConePngBase64(flavorName);

  const response = await renderStoreCardPng({
    storeName,
    flavorName,
    conePngBase64,
    accentColor,
  });

  const headers = new Headers(response.headers);
  for (const [k, v] of Object.entries(corsHeaders)) {
    headers.set(k, v);
  }
  headers.set('Cache-Control', 'public, max-age=86400');
  return new Response(response.body, { status: response.status, headers });
}
```
